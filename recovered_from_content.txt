598: class _TicketPainter extends CustomPainter {
599:   final Color color;
600:   
601:   _TicketPainter({required this.color});
602: 
603:   @override
604:   void paint(Canvas canvas, Size size) {
605:     final paint = Paint()..color = color;
606:     const double r = 12.0; // Corner radius for the black shape
607:     const double p = 8.0; // Orange border thickness
608:     
609:     // Intrusions based on proportions of the card
610:     final double tlW = size.width * 0.18;
611:     final double tlH = size.height * 0.35;
612:     
613:     final double blW = size.width * 0.48;
614:     final double blH = size.height * 0.40;
615: 
616:     final path = Path()
617:       // 1. Top edge, start after top-left intrusion
618:       ..moveTo(tlW + r, p)
619:       ..lineTo(size.width - p - r, p)
620:       
621:       // 2. Top-right corner (convex)
622:       ..arcToPoint(Offset(size.width - p, p + r), radius: const Radius.circular(r), clockwise: true)
623:       
624:       // 3. Right edge
625:       ..lineTo(size.width - p, size.height - p - r)
626:       
627:       // 4. Bottom-right corner (convex)
628:       ..arcToPoint(Offset(size.width - p - r, size.height - p), radius: const Radius.circular(r), clockwise: true)
629:       
630:       // 5. Bottom edge (right of bottom-left intrusion)
631:       ..lineTo(blW + r, size.height - p)
